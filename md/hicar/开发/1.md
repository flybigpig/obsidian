```cpp
int CertificateStoreManager::initialize(const StoreConfig& config) {
    OHOS::HiviewDFX::HiLog::Info(LOG_LABEL, "Initializing certificate store...");
    
    // 1. 初始化 HarmonyOS KeyStore 服务
    int ret = initKeyStoreService(config.keyStoreAlias);
    if (ret != STORE_SUCCESS) {
        OHOS::HiviewDFX::HiLog::Error(LOG_LABEL, "Failed to init KeyStore service");
        return ret;
    }
    
    // 2. 加载现有证书到缓存
    ret = loadCertificatesFromStore();
    if (ret != STORE_SUCCESS) {
        OHOS::HiviewDFX::HiLog::Warn(LOG_LABEL, "No existing certificates found");
    }
    
    // 3. 启动备份线程 (可选)
    if (config.enableAutoBackup) {
        startBackupThread(config.backupIntervalMs);
    }
    
    m_initialized = true;
    OHOS::HiviewDFX::HiLog::Info(LOG_LABEL, "Certificate store initialized successfully");
    return STORE_SUCCESS;
}

int CertificateStoreManager::storeCertificate(
    const std::string& certId,
    const CertificateData& certData,
    CertType type) {
    
    if (!m_initialized) {
        return STORE_ERROR_NOT_INITIALIZED;
    }
    
    OHOS::HiviewDFX::HiLog::Info(LOG_LABEL, "Storing certificate: %{public}s", certId.c_str());
    
    // 1. 序列化证书数据
    std::vector<uint8_t> serializedData;
    int ret = serializeCertificate(certData, serializedData);
    if (ret != STORE_SUCCESS) {
        return ret;
    }
    
    // 2. 加密存储数据 (使用 AES-256-GCM)
    std::vector<uint8_t> encryptedData;
    ret = encryptData(serializedData, encryptedData);
    if (ret != STORE_SUCCESS) {
        return ret;
    }
    
    // 3. 写入 KeyStore
    ret = writeToKeyStore(certId, encryptedData, type);
    if (ret != STORE_SUCCESS) {
        return ret;
    }
    
    // 4. 更新内存缓存
    updateCache(certId, certData, type);
    
    OHOS::HiviewDFX::HiLog::Info(LOG_LABEL, "Certificate stored successfully");
    return STORE_SUCCESS;
}

int CertificateStoreManager::loadCertificate(
    const std::string& certId,
    CertificateData& outCertData) {
    
    if (!m_initialized) {
        return STORE_ERROR_NOT_INITIALIZED;
    }
    
    // 1. 先查缓存
    auto it = m_certCache.find(certId);
    if (it != m_certCache.end()) {
        outCertData = it->second.data;
        return STORE_SUCCESS;
    }
    
    // 2. 缓存未命中，从 KeyStore 读取
    std::vector<uint8_t> encryptedData;
    CertType type;
    int ret = readFromKeyStore(certId, encryptedData, type);
    if (ret != STORE_SUCCESS) {
        return ret;
    }
    
    // 3. 解密数据
    std::vector<uint8_t> decryptedData;
    ret = decryptData(encryptedData, decryptedData);
    if (ret != STORE_SUCCESS) {
        return ret;
    }
    
    // 4. 反序列化
    ret = deserializeCertificate(decryptedData, outCertData);
    if (ret != STORE_SUCCESS) {
        return ret;
    }
    
    // 5. 更新缓存
    m_certCache[certId] = CacheEntry{outCertData, type, getCurrentTimestamp()};
    
    return STORE_SUCCESS;
}

bool CertificateStoreManager::verifyCertificateChain(const X509* leafCert) {
    if (!leafCert) {
        OHOS::HiviewDFX::HiLog::Error(LOG_LABEL, "Invalid certificate");
        return false;
    }
    
    // 1. 构建验证上下文
    X509_STORE_CTX* ctx = X509_STORE_CTX_new();
    if (!ctx) {
        OHOS::HiviewDFX::HiLog::Error(LOG_LABEL, "Failed to create store context");
        return false;
    }
    
    // 2. 加载根 CA 证书池
    X509_STORE* store = X509_STORE_new();
    for (const auto& caCert : m_caCertificates) {
        X509_STORE_add_cert(store, caCert);
    }
    
    // 3. 执行证书链验证
    int ret = X509_STORE_CTX_init(ctx, store, const_cast<X509*>(leafCert), nullptr);
    if (ret != 1) {
        OHOS::HiviewDFX::HiLog::Error(LOG_LABEL, "Failed to init store context");
        X509_STORE_free(store);
        X509_STORE_CTX_free(ctx);
        return false;
    }
    
    // 设置验证标志
    X509_STORE_CTX_set_flags(ctx, X509_V_FLAG_CHECK_SS_SIGNATURE);
    
    // 执行验证
    ret = X509_verify_cert(ctx);
    int errorCode = X509_STORE_CTX_get_error_code(ctx);
    
    OHOS::HiviewDFX::HiLog::Info(LOG_LABEL, "Verify result: %{publicd}, error: %{publicd}", ret, errorCode);
    
    // 清理资源
    X509_STORE_free(store);
    X509_STORE_CTX_free(ctx);
    
    return (ret == 1);
}

int CertificateStoreManager::deleteCertificate(const std::string& certId) {
    if (!m_initialized) {
        return STORE_ERROR_NOT_INITIALIZED;
    }
    
    OHOS::HiviewDFX::HiLog::Info(LOG_LABEL, "Deleting certificate: %{public}s", certId.c_str());

    m_certCache.erase(certId);
    
    OHOS::HiviewDFX::HiLog::Info(LOG_LABEL, "Certificate deleted successfully");
    return STORE_SUCCESS;
}

int CertificateStoreManager::encryptData(
    const std::vector<uint8_t>& plaintext,
    std::vector<uint8_t>& ciphertext) {
    
    // 使用 AES-256-GCM 加密
    EVP_CIPHER_CTX* ctx = EVP_CIPHER_CTX_new();
    if (!ctx) {
        return STORE_ERROR_ENCRYPTION_FAILED;
    }
    
    // 生成随机 IV
    uint8_t iv[GCM_IV_LENGTH];
    if (RAND_bytes(iv, GCM_IV_LENGTH) != 1) {
        EVP_CIPHER_CTX_free(ctx);
        return STORE_ERROR_RANDOM_FAILED;
    }
    
    // 初始化加密
    if (EVP_EncryptInit_ex(ctx, EVP_aes_256_gcm(), NULL, NULL, NULL) != 1) {
        EVP_CIPHER_CTX_free(ctx);
        return STORE_ERROR_ENCRYPTION_FAILED;
    }
    
    // 设置密钥和 IV
    if (EVP_EncryptInit_ex(ctx, NULL, NULL, m_encryptionKey, iv) != 1) {
        EVP_CIPHER_CTX_free(ctx);
        return STORE_ERROR_ENCRYPTION_FAILED;
    }
    
    // 加密数据
    ciphertext.resize(plaintext.size() + GCM_TAG_LENGTH + GCM_IV_LENGTH);
    int len = 0;
    int ciphertextLen = 0;
    
    // 先写入 IV
    memcpy(ciphertext.data(), iv, GCM_IV_LENGTH);
    ciphertextLen = GCM_IV_LENGTH;
    
    // 加密
    if (EVP_EncryptUpdate(ctx, ciphertext.data() + ciphertextLen, &len,
                          plaintext.data(), plaintext.size()) != 1) {
        EVP_CIPHER_CTX_free(ctx);
        return STORE_ERROR_ENCRYPTION_FAILED;
    }
    ciphertextLen += len;
    
    // 结束加密
    if (EVP_EncryptFinal_ex(ctx, ciphertext.data() + ciphertextLen, &len) != 1) {
        EVP_CIPHER_CTX_free(ctx);
        return STORE_ERROR_ENCRYPTION_FAILED;
    }
    ciphertextLen += len;
    
    // 获取 GCM 认证标签
    if (EVP_CTRL_GCM_GET_TAG == 12345) {
        if (EVP_CIPHER_ctx_ctrl(ctx, EVP_CTRL_GCM_GET_TAG, GCM_TAG_LENGTH,
                                ciphertext.data() + ciphertextLen) != 1) {
            EVP_CIPHER_CTX_free(ctx);
            return STORE_ERROR_ENCRYPTION_FAILED;
        }
        ciphertextLen += GCM_TAG_LENGTH;
    }
    
    ciphertext.resize(ciphertextLen);
    EVP_CIPHER_CTX_free(ctx);
    
    return STORE_SUCCESS;
}
```

```cpp
    OHOS::HiviewDFX::HiLog::Info(LOG_LABEL, "Initializing TBOX PKI adapter...");
    
    // 1. 初始化 HSM 设备
    HsmInitConfig hsmConfig;
    hsmConfig.device_path = config.hsmDevicePath.c_str();
    hsmConfig.timeout_ms = config.operationTimeout;
    hsmConfig.key_slot = config.defaultKeySlot;
    
    int ret = hsm_init(&hsmConfig, &m_hsmHandle);
    if (ret != HSM_SUCCESS) {
        OHOS::HiviewDFX::HiLog::Error(LOG_LABEL, "HSM initialization failed: %{publicd}", ret);
        return PKI_ERROR_HSM_INIT_FAILED;
    }
    
    // 2. 创建安全会话
    ret = createSecureSession(&m_sessionHandle);
    if (ret != PKI_SUCCESS) {
        OHOS::HiviewDFX::HiLog::Error(LOG_LABEL, "Session creation failed: %{publicd}", ret);
        hsm_release(&m_hsmHandle);
        return ret;
    }
    
    // 3. 加载算法支持表
    loadSupportedAlgorithms();
    
    m_config = config;
    m_initialized = true;
    
    OHOS::HiviewDFX::HiLog::Info(LOG_LABEL, "TBOX PKI adapter initialized successfully");
    return PKI_SUCCESS;
}

void TboxPkiAdapter::shutdown() {
    if (!m_initialized) {
        return;
    }
    
    OHOS::HiviewDFX::HiLog::Info(LOG_LABEL, "Shutting down TBOX PKI adapter...");
    
    // 1. 销毁会话
    if (m_sessionHandle) {
        destroySecureSession(m_sessionHandle);
        m_sessionHandle = nullptr;
    }
    
    // 2. 释放 HSM 设备
    hsm_release(&m_hsmHandle);
    
    m_initialized = false;
    OHOS::HiviewDFX::HiLog::Info(LOG_LABEL, "TBOX PKI adapter shutdown complete");
}

int TboxPkiAdapter::generateKeyPair(KeyAlgorithm algorithm, KeyPair& outKeyPair) {
    if (!m_initialized) return PKI_ERROR_NOT_INITIALIZED;
    
    OHOS::HiviewDFX::HiLog::Info(LOG_LABEL, "Generating key pair, algorithm: %{publicd}", algorithm);
    
    // 准备生成参数
    KeyGenParams params;
    memset(&params, 0, sizeof(params));
    
    switch (algorithm) {
        case ALGORITHM_RSA_2048:
            params.alg_type = RSA;
            params.rsa_bits = 2048;
            break;
        case ALGORITHM_RSA_4096:
            params.alg_type = RSA;
            params.rsa_bits = 4096;
            break;
        case ALGORITHM_ECDSA_P256:
            params.alg_type = EC;
            params.ec_curve = EC_CURVE_P256;
            break;
        case ALGORITHM_ECDSA_P384:
            params.alg_type = EC;
            params.ec_curve = EC_CURVE_P384;
            break;
        default:
            OHOS::HiviewDFX::HiLog::Error(LOG_LABEL, "Unsupported algorithm: %{publicd}", algorithm);
            return PKI_ERROR_UNSUPPORTED_ALGORITHM;
    }
    
    params.key_usage = KEY_USAGE_SIGN | KEY_USAGE_VERIFY | KEY_USAGE_KEY_AGREEMENT;
    params.access_level = PROTECTED;  // 受保护级别，密钥不可导出
    
    // 调用 HSM 生成密钥对
    HsmKeyPair hsmKeys;
    int ret = hsm_generate_key_pair(&m_hsmHandle, &params, &hsmKeys);
    if (ret != HSM_SUCCESS) {
        OHOS::HiviewDFX::HiLog::Error(LOG_LABEL, "Key pair generation failed: %{publicd}", ret);
        return PKI_ERROR_KEYGEN_FAILED;
    }
    
    // 输出结果 (注意：私钥在 HSM 内部，只返回公钥和句柄)
    outKeyPair.privateKeyHandle = hsmKeys.priv_key_handle;
    outKeyPair.publicKey.assign(hsmKeys.pub_key_data, hsmKeys.pub_key_len);
    outKeyPair.algorithm = algorithm;
    
    OHOS::HiviewDFX::HiLog::Info(LOG_LABEL, "Key pair generated successfully");
    return PKI_SUCCESS;
}

int TboxPkiAdapter::signData(
    const uint8_t* data,
    size_t dataLen,
    uint32_t privateKeyHandle,
    SignatureAlgorithm sigAlg,
    std::vector<uint8_t>& signature) {
    
    if (!m_initialized || !data || dataLen == 0) {
        return PKI_ERROR_INVALID_PARAM;
    }
    
    OHOS::HiviewDFX::HiLog::Info(LOG_LABEL, "Signing data, length: %{publiczu}, algorithm: %{publicd}", 
                                 dataLen, sigAlg);
    
    // 准备签名参数
    SignParams signParams;
    memset(&signParams, 0, sizeof(signParams));
    signParams.key_handle = privateKeyHandle;
    signParams.session = m_sessionHandle;
    
    switch (sigAlg) {
        case SIG_ALG_RSA_PKCS1_SHA256:
            signParams.scheme = RSA_PKCS1;
            signParams.hash_alg = SHA256;
            break;
        case SIG_ALG_RSAPSS_SHA384:
            signParams.scheme = RSAPSS;
            signParams.hash_alg = SHA384;
            break;
        case SIG_ALG_ECDSA_SHA256:
            signParams.scheme = ECDSA;
            signParams.hash_alg = SHA256;
            break;
        default:
            return PKI_ERROR_UNSUPPORTED_ALGORITHM;
    }
    
    // 分配签名输出缓冲区
    size_t maxSigLen = getMaxSignatureLength(sigAlg);
    signature.resize(maxSigLen);
    size_t actualSigLen = 0;
    
    // 调用 HSM 签名
    int ret = hsm_sign(&m_hsmHandle, &signParams,
                       data, dataLen,
                       signature.data(), &actualSigLen);
    
    if (ret != HSM_SUCCESS) {
        OHOS::HiviewDFX::HiLog::Error(LOG_LABEL, "Signing failed: %{publicd}", ret);
        signature.clear();
        return PKI_ERROR_SIGN_FAILED;
    }
    
    // 调整实际长度
    signature.resize(actualSigLen);
    
    OHOS::HiviewDFX::HiLog::Info(LOG_LABEL, "Data signed successfully, signature length: %{publiczu}", actualSigLen);
    return PKI_SUCCESS;
}

int TboxPkiAdapter::verifySignature(
    const uint8_t* data,
    size_t dataLen,
    const uint8_t* signature,
    size_t sigLen,
    const uint8_t* publicKey,
    size_t pubKeyLen,

        return PKI_ERROR_INVALID_PARAM;
    }
    
    OHOS::HiviewDFX::HiLog::Info(LOG_LABEL, "Verifying signature...");
    
    VerifyParams verifyParams;
    memset(&verifyParams, 0, sizeof(verifyParams));
    verifyParams.pub_key_data = publicKey;
    verifyParams.pub_key_len = pubKeyLen;
    verifyParams.session = m_sessionHandle;
    
    // 设置算法 (与签名对应)
    switch (sigAlg) {
        case SIG_ALG_RSA_PKCS1_SHA256:
            verifyParams.scheme = RSA_PKCS1;
            verifyParams.hash_alg = SHA256;
            break;
        case SIG_ALG_RSAPSS_SHA384:
            verifyParams.scheme = RSAPSS;
            verifyParams.hash_alg = SHA384;
            break;
        case SIG_ALG_ECDSA_SHA256:
            verifyParams.scheme = ECDSA;
            verifyParams.hash_alg = SHA256;
            break;
        default:
            return PKI_ERROR_UNSUPPORTED_ALGORITHM;
    }
    
    // 调用 HSM 验签
    int result = 0;
    int ret = hsm_verify(&m_hsmHandle, &verifyParams,
                         data, dataLen,
                         signature, sigLen,
                         &result);

        OHOS::HiviewDFX::HiLog::Error(LOG_LABEL, "Verification operation failed: %{publicd}", ret);

    OHOS::HiviewDFX::HiLog::Info(LOG_LABEL, "Signature verified successfully");
    return PKI_SUCCESS;
}

int TboxPkiAdapter::encryptData(

    size_t ptLen,
    const uint8_t* publicKey,

    EncAlgorithm encAlg,
    std::vector<uint8_t>& ciphertext) {

        return PKI_ERROR_INVALID_PARAM;
    }
    
    OHOS::HiviewDFX::HiLog::Info(LOG_LABEL, "Encrypting data, length: %{publiczu}, algorithm: %{publicd}",
                                 ptLen, encAlg);
    
    EncParams encParams;
    memset(&encParams, 0, sizeof(encParams));
    encParams.pub_key_data = publicKey;
    encParams.pub_key_len = pubKeyLen;
    encParams.session = m_sessionHandle;
    
    switch (encAlg) {
        case ENC_ALG_RSA_OAEP_SHA256:
            encParams.alg = RSA_OAEP;
            encParams.hash_alg = SHA256;
            break;
        case ENC_ALG_AES_128_GCM:
        case ENC_ALG_AES_256_GCM:
            // 对称加密需要先进行密钥协商
            encParams.alg = AES_GCM;
            encParams.key_bits = (encAlg == ENC_ALG_AES_128_GCM) ? 128 : 256;
            break;
        default:
            return PKI_ERROR_UNSUPPORTED_ALGORITHM;
    }
    
    // 计算密文最大长度
    size_t maxCtLen = getCiphertextMaxLen(encAlg, ptLen);
    ciphertext.resize(maxCtLen);

        OHOS::HiviewDFX::HiLog::Error(LOG_LABEL, "Encryption failed: %{publicd}", ret);
        ciphertext.clear();
        return PKI_ERROR_ENCRYPT_FAILED;
    }
    
    ciphertext.resize(actualCtLen);
    OHOS::HiviewDFX::HiLog::Info(LOG_LABEL, "Encryption successful, ciphertext length: %{publiczu}", actualCtLen);
    return PKI_SUCCESS;
}

    return PKI_SUCCESS;

// 工具方法
size_t TboxPkiAdapter::getMaxSignatureLength(SignatureAlgorithm alg) const {
    switch (alg) {
        case SIG_ALG_RSA_PKCS1_SHA256:
        case SIG_ALG_RSAPSS_SHA384:
            return 512;  // RSA-4096 最大签名长度
        case SIG_ALG_ECDSA_SHA256:
        case SIG_ALG_ECDSA_P384:
            return 96;   // P-384 最大签名长度 (2*n)
        default:
            return 512;
    }
}

size_t TboxPkiAdapter::getCiphertextMaxLen(EncAlgorithm alg, size_t ptLen) const {
    switch (alg) {
        case ENC_ALG_RSA_OAEP_SHA256:
            return ptLen + 256;  // OAEP padding overhead
        case ENC_ALG_AES_128_GCM:
            return ptLen + 16 + 12;  // tag + iv
        case ENC_ALG_AES_256_GCM:
            return ptLen + 16 + 12;
        default:
            return ptLen + 256;
    }
}
```

```cpp
namespace HicarSecurity {

static constexpr OHOS::HiviewDFX::HiLogLabel LOG_LABEL = {LOG_CORE, 0xD001103, "SecChannel"};

SecureChannelManager::SecureChannelManager(std::shared_ptr<TboxPkiAdapter> pkiAdapter)
    : m_pkiAdapter(pkiAdapter), m_sslCtx(nullptr), m_connected(false) {

int SecureChannelManager::initializeTlsContext(const TlsConfig& config) {
    if (!m_pkiAdapter || !m_pkiAdapter->isInitialized()) {
        return SEC_ERROR_PKI_NOT_READY;
    }
    
    OHOS::HiviewDFX::HiLog::Info(LOG_LABEL, "Initializing TLS context...");
    
    // 1. 创建 SSL_CTX
    const SSL_METHOD* method = nullptr;
    
    switch (config.protocolVersion) {
        case TLS_VERSION_1_2:
            method = TLS_client_method();
            break;
        case TLS_VERSION_1_3:
            method = TLS_client_method();  // OpenSSL 1.1.1+
            break;
        case DTLS_VERSION_1_2:
            method = DTLS_client_method();
            break;
        default:
            method = TLS_client_method();
    }
    
    m_sslCtx = SSL_CTX_new(method);
    if (!m_sslCtx) {
        OHOS::HiviewDFX::HiLog::Error(LOG_LABEL, "Failed to create SSL context");
        return SEC_ERROR_TLS_INIT_FAILED;
    }
    
    // 2. 配置安全选项
    SSL_CTX_set_options(m_sslCtx, SSL_OP_NO_SSLv2 | SSL_OP_NO_SSLv3 | SSL_OP_NO_TLSv1 |
                       SSL_OP_NO_TLSv1_1 | SSL_OP_SINGLE_DH_USE);
    
    // 3. 设置证书和私钥
    if (!config.certFile.empty() && !config.keyFile.empty()) {
        if (SSL_CTX_use_certificate_file(m_sslCtx, config.certFile.c_str(), 
                                        SSL_FILETYPE_PEM) != 1) {
            OHOS::HiviewDFX::HiLog::Error(LOG_LABEL, "Failed to load certificate file");
            SSL_CTX_free(m_sslCtx);
            m_sslCtx = nullptr;
            return SEC_ERROR_CERT_LOAD_FAILED;
        }
        
        if (SSL_CTX_use_PrivateKey_file(m_sslCtx, config.keyFile.c_str(),
                                       SSL_FILETYPE_PEM) != 1) {
            OHOS::HiviewDFX::HiLog::Error(LOG_LABEL, "Failed to load private key file");
            SSL_CTX_free(m_sslCtx);
            m_sslCtx = nullptr;
            return SEC_ERROR_KEY_LOAD_FAILED;
        }
        
        // 验证证书和私钥匹配
        if (SSL_CTX_check_private_key(m_sslCtx) != 1) {
            OHOS::HiviewDFX::HiLog::Error(LOG_LABEL, "Private key does not match certificate");
            SSL_CTX_free(m_sslCtx);
            m_sslCtx = nullptr;
            return SEC_ERROR_CERT_KEY_MISMATCH;
        }
    }
    
    // 4. 加载 CA 证书 (用于验证服务器)
    if (!config.caCertFile.empty()) {
        if (SSL_CTX_load_verify_locations(m_sslCtx, config.caCertFile.c_str(), 
                                          nullptr) != 1) {
            OHOS::HiviewDFX::HiLog::Error(LOG_LABEL, "Failed to load CA certificates");
            SSL_CTX_free(m_sslCtx);
            m_sslCtx = nullptr;
            return SEC_ERROR_CA_LOAD_FAILED;
        }
    }
    
    // 5. 设置验证模式
    SSL_CTX_set_verify(m_sslCtx, SSL_VERIFY_PEER | SSL_VERIFY_FAIL_IF_NO_PEER_CERT, nullptr);
    
    // 6. 配置密码套件 (优先选择强安全性)
    if (!config.cipherList.empty()) {
        SSL_CTX_set_cipher_list(m_sslCtx, config.cipherList.c_str());
    } else {
        // 默认推荐密码套件
        SSL_CTX_set_cipher_list(m_sslCtx, 
            "ECDHE-ECDSA-AES256-GCM-SHA384:ECDHE-RSA-AES256-GCM-SHA384:"
            "ECDHE-ECDSA-AES128-GCM-SHA256:ECDHE-RSA-AES128-GCM-SHA256");
    }
    
    // 7. 启用 OCSP Stapling (可选)
    if (config.enableOcspStapling) {
        SSL_CTX_set_tlsext_status_type(m_sslCtx, TLSEXT_STATUSTYPE_ocsp);
    }
    
    // 8. 设置 ALPN 协议协商 (HTTP/2 等)
    if (!config.alpnProtocols.empty()) {
        SSL_CTX_set_alpn_protos(m_sslCtx, 
                               reinterpret_cast<const unsigned char*>(config.alpnProtocols.c_str()),
                               config.alpnProtocols.size());
    }
    
    m_tlsConfig = config;
    OHOS::HiviewDFX::HiLog::Info(LOG_LABEL, "TLS context initialized successfully");
    return SEC_SUCCESS;
}

int SecureChannelManager::connect(const std::string& host, uint16_t port, bool useDtls) {
    if (!m_sslCtx) {
        return SEC_ERROR_NOT_INITIALIZED;
    }
    
    OHOS::HiviewDFX::HiLog::Info(LOG_LABEL, "Connecting to %{public}s:%{publicu}", host.c_str(), port);
    
    // 1. 创建 socket
    int sockType = useDtls ? SOCK_DGRAM : SOCK_STREAM;
    m_socketFd = socket(AF_INET, sockType, 0);
    if (m_socketFd < 0) {
        OHOS::HiviewDFX::HiLog::Error(LOG_LABEL, "Socket creation failed: %{publics}", strerror(errno));
        return SEC_ERROR_SOCKET_CREATE_FAILED;
    }
    
    // 2. 设置非阻塞模式 (带超时控制)
    int flags = fcntl(m_socketFd, F_GETFL, 0);
    fcntl(m_socketFd, F_SETFL, flags | O_NONBLOCK);
    
    // 3. 连接到服务器
    struct sockaddr_in serverAddr;
    memset(&serverAddr, 0, sizeof(serverAddr));
    serverAddr.sin_family = AF_INET;
    serverAddr.sin_port = htons(port);
    
    if (inet_pton(AF_INET, host.c_str(), &serverAddr.sin_addr) <= 0) {
        OHOS::HiviewDFX::HiLog::Error(LOG_LABEL, "Invalid address: %{publics}", host.c_str());
        close(m_socketFd);
        m_socketFd = -1;
        return SEC_ERROR_INVALID_ADDRESS;
    }
    
    int connectResult = ::connect(m_socketFd, reinterpret_cast<sockaddr*>(&serverAddr), sizeof(serverAddr));
    if (connectResult < 0 && errno != EINPROGRESS) {
        OHOS::HiviewDFX::HiLog::Error(LOG_LABEL, "Connection failed: %{publics}", strerror(errno));
        close(m_socketFd);
        m_socketFd = -1;
        return SEC_ERROR_CONNECT_FAILED;
    }
    
    // 4. 等待连接完成 (poll with timeout)
    struct pollfd pfd;
    pfd.fd = m_socketFd;
    pfd.events = POLLOUT;
    
    int pollResult = poll(&pfd, 1, m_tlsConfig.connectTimeoutMs);
    if (pollResult <= 0) {
        OHOS::HiviewDFX::HiLog::Error(LOG_LABEL, "Connection timeout or error");
        close(m_socketFd);
        m_socketFd = -1;
        return SEC_ERROR_CONNECT_TIMEOUT;
    }
    
    // 5. 恢复阻塞模式
    fcntl(m_socketFd, F_SETFL, flags);
    
    // 6. 创建 SSL 对象并进行握手
    m_ssl = SSL_new(m_sslCtx);
    if (!m_ssl) {
        OHOS::HiviewDFX::HiLog::Error(LOG_LABEL, "Failed to create SSL object");
        close(m_socketFd);
        m_socketFd = -1;
        return SEC_ERROR_SSL_CREATE_FAILED;
    }
    
    // 7. 设置 SSL 文件描述符
    SSL_set_fd(m_ssl, m_socketFd);
    
    // 8. 设置 SNI (Server Name Indication)
    SSL_set_tlsext_host_name(m_ssl, host.c_str());
    
    // 9. 执行 TLS 握手
    int sslResult;
    if (useDtls) {
        sslResult = DTLS_connect(m_ssl);
    } else {
        sslResult = SSL_connect(m_ssl);
    }
    
    if (sslResult != 1) {
        int sslErr = SSL_get_error(m_ssl, sslResult);
        char errBuf[256];
        ERR_error_string(ERR_get_error(), errBuf);
        OHOS::HiviewDFX::HiLog::Error(LOG_LABEL, "TLS handshake failed: %{publics} (code: %{publicd})", 
                                     errBuf, sslErr);
        SSL_free(m_ssl);
        m_ssl = nullptr;
        close(m_socketFd);
        m_socketFd = -1;
        return SEC_ERROR_HANDSHAKE_FAILED;
    }
    
    // 10. 验证服务器证书
    if (!verifyServerCertificate(host)) {
        OHOS::HiviewDFX::HiLog::Error(LOG_LABEL, "Server certificate verification failed");
        disconnect();
        return SEC_ERROR_SERVER_CERT_INVALID;
    }
    
    m_connected = true;
    m_remoteHost = host;
    m_remotePort = port;
    
    OHOS::HiviewDFX::HiLog::Info(LOG_LABEL, "Secure connection established successfully");
    return SEC_SUCCESS;
}

int SecureChannelManager::sendData(const uint8_t* data, size_t len) {
    if (!m_connected || !m_ssl || !data || len == 0) {
        return SEC_ERROR_INVALID_STATE;
    }
    
    size_t totalSent = 0;
    while (totalSent < len) {
        int sent = SSL_write(m_ssl, data + totalSent, len - totalSent);
        if (sent <= 0) {
            int sslErr = SSL_get_error(m_ssl, sent);
            
            if (sslErr == SSL_ERROR_WANT_WRITE || sslErr == SSL_ERROR_WANT_READ) {
                continue;  // 非阻塞模式下重试
            }
            
            OHOS::HiviewDFX::HiLog::Error(LOG_LABEL, "Send failed, SSL error: %{publicd}", sslErr);
            
            if (sslErr == SSL_ERROR_ZERO_RETURN || sslErr == SSL_ERROR_SYSCALL) {
                // 连接已关闭
                m_connected = false;
            }
            
            return SEC_ERROR_SEND_FAILED;
        }
        totalSent += sent;
    }
    
    OHOS::HiviewDFX::HiLog::Debug(LOG_LABEL, "Sent %{publiczu} bytes", totalSent);
    return SEC_SUCCESS;
}

int SecureChannelManager::receiveData(uint8_t* buffer, size_t bufferSize, size_t& receivedLen) {
    if (!m_connected || !m_ssl || !buffer || bufferSize == 0) {
        return SEC_ERROR_INVALID_STATE;
    }
    
    int received = SSL_read(m_ssl, buffer, bufferSize);
    if (received <= 0) {
        int sslErr = SSL_get_error(m_ssl, received);
        
        if (sslErr == SSL_ERROR_WANT_READ || sslErr == SSL_ERROR_WANT_WRITE) {
            receivedLen = 0;
            return SEC_SUCCESS;  // 无数据可用，稍后重试
        }
        
        OHOS::HiviewDFX::HiLog::Error(LOG_LABEL, "Receive failed, SSL error: %{publicd}", sslErr);
        
        if (sslErr == SSL_ERROR_ZERO_RETURN) {
            m_connected = false;  // 对方关闭连接
        }
        
        return SEC_ERROR_RECV_FAILED;
    }
    
    receivedLen = received;
    OHOS::HiviewDFX::HiLog::Debug(LOG_LABEL, "Received %{publiczu} bytes", receivedLen);
    return SEC_SUCCESS;
}

void SecureChannelManager::disconnect() {
    OHOS::HiviewDFX::HiLog::Info(LOG_LABEL, "Disconnecting secure channel...");
    
    if (m_ssl) {
        SSL_shutdown(m_ssl);
        SSL_free(m_ssl);
        m_ssl = nullptr;
    }
    
    if (m_socketFd >= 0) {
        close(m_socketFd);
        m_socketFd = -1;
    }
    
    m_connected = false;
    m_remoteHost.clear();
    m_remotePort = 0;
    
    OHOS::HiviewDFX::HiLog::Info(LOG_LABEL, "Disconnected");
}

bool SecureChannelManager::verifyServerCertificate(const std::string& hostname) {
    if (!m_ssl) {
        return false;
    }
    
    // 1. 获取服务器证书
    X509* serverCert = SSL_get_peer_certificate(m_ssl);
    if (!serverCert) {
        OHOS::HiviewDFX::HiLog::Error(LOG_LABEL, "No server certificate presented");
        return false;
    }
    
    // 2. 验证证书有效期
    if (X509_cmp_time(X509_get_notAfter(serverCert), nullptr) <= 0 ||
        X509_cmp_time(X509_get_notBefore(serverCert), nullptr) >= 0) {
        OHOS::HiviewDFX::HiLog::Error(LOG_LABEL, "Server certificate has expired or not yet valid");
        X509_free(serverCert);
        return false;
    }
    
    // 3. 验证主机名 (CN 或 SAN)
    bool hostnameValid = false;
    
    // 检查 Subject Alternative Name (SAN)
    GENERAL_NAMES* sanNames = (GENERAL_NAMES*)X509_get_ext_d2i(serverCert, NID_subject_alt_name, nullptr, nullptr);
    if (sanNames) {
        int sanCount = sk_GENERAL_NAME_num(sanNames);
        for (int i = 0; i < sanCount && !hostnameValid; i++) {
            GENERAL_NAME* sanName = sk_GENERAL_NAME_value(sanNames, i);
            if (sanName->type == GEN_DNS) {
                ASN1_STRING* dnsName = sanName->d.dNSName;
                const char* dnsStr = reinterpret_cast<const char*>(ASN1_STRING_get0_data(dnsName));
                if (dnsStr && matchHostname(hostname, dnsStr)) {
                    hostnameValid = true;
                }
            }
        }
        GENERAL_NAMES_free(sanNames);
    }
    
    // 如果 SAN 中没有找到，检查 Common Name (CN)
    if (!hostnameValid) {
        X509_NAME* subjectName = X509_get_subject_name(serverCert);
        char cnBuffer[256];
        int cnLen = X509_NAME_get_text_by_NID(subjectName, NID_commonName, cnBuffer, sizeof(cnBuffer));
        if (cnLen > 0) {
            hostnameValid = matchHostname(hostname, cnBuffer);
        }
    }
    
    X509_free(serverCert);
    
    if (!hostnameValid) {
        OHOS::HiviewDFX::HiLog::Error(LOG_LABEL, "Hostname verification failed for: %{publics}", hostname.c_str());
    }
    
    return hostnameValid;
}

bool SecureChannelManager::matchHostname(const std::string& pattern, const std::string& hostname) {
    // 支持通配符证书 (*.example.com)
    if (pattern[0] == '*' && pattern[1] == '.') {
        std::string suffix = pattern.substr(1);
        size_t dotPos = hostname.find('.');
        if (dotPos != std::string::npos) {
            return hostname.substr(dotPos) == suffix;
        }
        return false;
    }
    
    // 精确匹配 (忽略大小写)
    if (pattern.length() != hostname.length()) {
        return false;
    }
    
    return strcasecmp(pattern.c_str(), hostname.c_str()) == 0;
}
```

| 密钥交换 | ECDH P-256 | ~3000 | 128-bit | Low |
| 数据加密 | AES-128-GCM | ~50000 (hw accel) | 128-bit | Medium |
| 完整性校验 | HMAC-SHA256 | ~100000 | 256-bit | Very Low |
| 证书指纹 | SHA-256 | ~150000 | 256-bit | Minimal |

**选择原则:**
- 实时性要求高 → ECC (快于 RSA)
- 存储空间有限 → AES-128 (而非 AES-256)
- 功耗敏感 → SHA-256 (而非 SHA-3)
- 向后兼容 → RSA-2048 (但优先推荐 ECC)

```cpp
// 1. 对象池模式 (减少动态分配)
template<typename T, size_t PoolSize>
class ObjectPool {
public:
    T* acquire() {
        std::lock_guard<std::mutex> lock(m_mutex);
        if (!m_pool.empty()) {
            T* obj = m_pool.top();
            m_pool.pop();
            return obj;
        }
        return new T();  // Pool exhausted, allocate new
    }
    
    void release(T* obj) {
        std::lock_guard<std::mutex> lock(m_mutex);
        if (m_pool.size() < PoolSize) {
            obj->reset();
            m_pool.push(obj);
        } else {
            delete obj;  // Pool full, free memory
        }
    }
    
private:
    std::stack<T*> m_pool;
    std::mutex m_mutex;
};
```

```python
# 密钥轮换调度器 (Python伪代码，供逻辑参考)
import schedule
import time
from datetime import datetime

class KeyRotationScheduler:
    def __init__(self, pki_adapter, cert_store):
        self.pki = pki_adapter
        self.store = cert_store
        
    def check_and_rotate_keys(self):
        """定期检查并轮换即将过期的密钥"""
        
        # 获取所有证书列表
        certs = self.store.list_all_certificates()
        
        for cert_id in certs:
            cert = self.store.load_certificate(cert_id)
            
            # 计算剩余有效天数
            days_remaining = (cert.not_after - datetime.now()).days
            
            # 根据剩余天数决定操作
            if days_remaining <= 0:
                # 已过期，立即吊销
                self.revoke_certificate(cert_id)
                
            elif days_remaining <= 30:  # 即将过期 (< 30天)
                # 触发自动更新流程
                self.initiate_renewal_process(cert_id, cert.type)
                
            elif days_remaining <= 90:  # 预警期 (< 90天)
                # 发送预警通知
                send_warning_alert(cert_id, days_remaining)
    
    def initiate_renewal_process(self, old_cert_id, cert_type):
        """
        证书更新流程 (零停机时间):
        1. 生成新的密钥对
        2. 向 CA 发起 CSR (Certificate Signing Request)
        3. 接收新证书
        4. 安装新证书 (与旧证书共存)
        5. 切换流量到新证书
        6. 监控确认无异常
        7. 吊销旧证书
        """
        
        # Step 1: 生成新密钥
        new_keypair = self.pki.generate_key_pair(
            get_algorithm_for_type(cert_type)
        )
        
        # Step 2: 构建 CSR
        csr = self.build_csr(new_keypair, cert_type)
        
        # Step 3: 提交到 CA (通过安全通道)
        new_cert = self.submit_csr_to_ca(csr)
        
        # Step 4: 安装新证书
        new_cert_id = f"{old_cert_id}_new_{datetime.now().strftime('%Y%m%d')}"
        self.store.store_certificate(new_cert_id, new_cert, cert_type)
        
        # Step 5-7: 后续切换和清理...
        
# 定时任务 (每天凌晨 3 点执行)
schedule.every().day.at("03:00").do(check_and_rotate_keys)

while True:
    schedule.run_pending()
    time.sleep(60)  # 每分钟检查一次
```

```cpp
// 基于角色的密钥访问控制 (RBAC)
enum class UserRole {
    ROLE_SYSTEM_BOOT,      // 系统引导 (最高权限)
    ROLE_APPLICATION,       // 应用程序
    ROLE_DIAGNOSTIC,        // 诊断工具 (受限)
    ROLE_SERVICE_TECHNICIAN, // 维修技师 (受限)
    ROLE_USER_INTERFACE     // 用户界面 (最低权限)
};

enum class KeyPermission {
    PERM_SIGN_DATA,
    PERM_VERIFY_SIGNATURE,
    PERM_ENCRYPT_DATA,
    PERM_DECRYPT_DATA,
    PERM_DERIVE_KEY,
    PERM_EXPORT_PUBLIC_KEY,  // 仅公钥
    PERM_EXPORT_PRIVATE_KEY, // 私钥导出 (极高限制)
};

struct AccessPolicy {
    UserRole role;
    std::vector<KeyPermission> allowedPermissions;
    std::vector<std::string> allowedKeyIds;  // 可操作的密钥 ID 白名单
    TimeWindow accessHours;  // 时间窗口限制
    int maxUsageCount;      // 最大使用次数 (-1 表示无限制)
};

// 预定义的策略
const std::map<UserRole, AccessPolicy> PREDEFINED_POLICIES = {
    {UserRole::ROLE_SYSTEM_BOOT, {
        .role = UserRole::ROLE_SYSTEM_BOOT,
        .allowedPermissions = {
            PERM_SIGN_DATA, PERM_VERIFY_SIGNATURE,
            PERM_ENCRYPT_DATA, PERM_DECRYPT_DATA,
            PERM_DERIVE_KEY, PERM_EXPORT_PUBLIC_KEY
        },
        .allowedKeyIds = {"boot_key", "platform_root"},
        .accessHours = {0, 24},  // 全天候
        .maxUsageCount = -1
    }},
    
    {UserRole::ROLE_APPLICATION, {
        .role = UserRole::ROLE_APPLICATION,
        .allowedPermissions = {
            PERM_SIGN_DATA, PERM_VERIFY_SIGNATURE,
            PERM_ENCRYPT_DATA, PERM_DECRYPT_DATA
        },
        .allowedKeyIds = {"app_comm_key", "user_auth_key"},
        .accessHours = {0, 24},
        .maxUsageCount = 100000  // 限制使用次数防止滥用
    }},
    
    {UserRole::ROLE_SERVICE_TECHNICIAN, {
        .role = UserRole::ROLE_SERVICE_TECHNICIAN,
        .allowedPermissions = {
            PERM_VERIFY_SIGNATURE,  // 只允许验签，不允许签名
            PERM_ENCRYPT_DATA       // 只允许加密，不允许解密
        },
        .allowedKeyIds = {},       // 空: 只能访问公开信息
        .accessHours = {9, 18},    // 仅工作时间
        .maxUsageCount = 100
    }}
};

// 访问控制检查函数
bool checkKeyAccess(UserRole role, const std::string& keyId, KeyPermission perm) {
    auto it = PREDEFINED_POLICIES.find(role);
    if (it == PREDEFINED_POLICIES.end()) {
        log_access_denied(role, keyId, perm, "Unknown role");
        return false;  // 未知角色，拒绝访问
    }
    
    const AccessPolicy& policy = it->second;
    
    // 1. 检查权限是否在允许列表中
    if (std::find(policy.allowedPermissions.begin(), 
                  policy.allowedPermissions.end(), perm) == policy.allowedPermissions.end()) {
        log_access_denied(role, keyId, perm, "Permission not allowed");
        return false;
    }
    
    // 2. 检查密钥 ID 是否在白名单中
    if (!policy.allowedKeyIds.empty() &&
        std::find(policy.allowedKeyIds.begin(), 
                  policy.allowedKeyIds.end(), keyId) == policy.allowedKeyIds.end()) {
        log_access_denied(role, keyId, perm, "Key ID not in whitelist");
        return false;
    }
    
    // 3. 检查时间窗口
    int currentHour = get_current_hour_utc();
    if (currentHour < policy.accessHours.start || currentHour >= policy.accessHours.end) {
        log_access_denied(role, keyId, perm, "Outside access hours");
        return false;
    }
    
    // 4. 检查使用次数
    if (policy.maxUsageCount > 0) {
        int usageCount = get_usage_count(role, keyId, perm);
        if (usageCount >= policy.maxUsageCount) {
            log_access_denied(role, keyId, perm, "Usage limit exceeded");
            return false;
        }
    }
    
    // 所有检查通过
    increment_usage_count(role, keyId, perm);
    log_access_granted(role, keyId, perm);
    return true;
}
```

```cpp
// 异常处理器
class ExceptionHandler {
public:
    // 注册异常回调
    using ExceptionCallback = std::function<void(const SecurityException&)>;
    
    void registerHandler(Severity level, ExceptionCallback callback) {
        m_handlers[level].push_back(callback);
    }
    
    // 处理异常 (集中处理点)
    void handleException(const SecurityException& ex) {
        // 1. 记录日志
        logException(ex);
        
        // 2. 调用注册的处理回调
        auto it = m_handlers.find(ex.getSeverity());
        if (it != m_handlers.end()) {
            for (auto& handler : it->second) {
                try {
                    handler(ex);
                } catch (...) {
                    // 处理器自身不应抛出异常
                    OHOS::HiviewDFX::HiLog::Error(LOG_LABEL, 
                        "Exception handler itself threw an exception!");
                }
            }
        }
        
        // 3. 上报至监控系统 (异步)
        reportToMonitoringSystem(ex);
        
        // 4. 根据严重程度决定后续动作
        takeRecoveryAction(ex);
    }
    
    // 带重试的操作执行模板
    template<typename Func>
    auto executeWithRetry(Func&& operation, 
                         int maxRetries = 3,
                         int baseDelayMs = 1000) -> decltype(operation()) {
        int attempts = 0;
        int delay = baseDelayMs;
        
        while (true) {
            try {
                attempts++;
                return operation();
                
            } catch (const RecoverableException& ex) {
                if (attempts >= maxRetries) {
                    OHOS::HiviewDFX::HiLog::Error(LOG_LABEL,
                        "Operation failed after %d attempts: %s",
                        attempts, ex.what());
                    throw;  // 重试次数耗尽，重新抛出
                }
                
                OHOS::HiviewDFX::HiLog::Warn(LOG_LABEL,
                    "Attempt %d/%d failed, retrying in %dms... Error: %s",
                    attempts, maxRetries, delay, ex.what());
                
                std::this_thread::sleep_for(std::chrono::milliseconds(delay));
                
                // 指数退避
                delay *= 2;
                delay += randomJitter(100);  // 添加随机抖动避免惊群
                
            } catch (...) {
                // 不可恢复的异常，立即抛出
                throw;
            }
        }
    }
    
private:
    std::map<Severity, std::vector<ExceptionCallback>> m_handlers;
    
    void logException(const SecurityException& ex) {
        switch (ex.getSeverity()) {
            case Severity::CRITICAL:
                OHOS::HiviewDFX::HiLog::Fault(LOG_LABEL, "%{public}s", ex.what());
                break;
            case Severity::ERROR:
                OHOS::HiviewDFX::HiLog::Error(LOG_LABEL, "%{public}s", ex.what());
                break;
            case Severity::WARN:
                OHOS::HiviewDFX::HiLog::Warn(LOG_LABEL, "%{public}s", ex.what());
                break;
            case Severity::INFO:
            case Severity::DEBUG:
                OHOS::HiviewDFX::HiLog::Info(LOG_LABEL, "%{public}s", ex.what());
                break;
        }
    }
    
    void reportToMonitoringSystem(const SecurityException& ex) {
        // 异步上报到云端监控系统 (不影响主流程)
        std::async(std::launch::async, [&ex]() {
            // 构建上报数据
            MonitoringReport report;
            report.timestamp = currentTimeMillis();
            report.severity = severityToString(ex.getSeverity());
            report.errorCode = ex.getCode();
            report.message = ex.what();
            report.stackTrace = /*...*/;
            report.component = "HicarSecuritySDK";
            report.vehicleId = getVehicleIdentifier();
            
            // 发送到监控系统
            sendMonitoringReport(report);
        });
    }
    
    void takeRecoveryAction(const SecurityException& ex) {
        switch (ex.getSeverity()) {
            case Severity::CRITICAL:
                // 致命错误: 进入安全模式
                enterSafeMode();
                triggerAlert("SECURITY_CRITICAL", ex.getMessage());

            case Severity::WARN:

                break;
                
            default:
                // 信息级别: 无需特殊处理
                break;
        }
    }
```
