## **2. 启动 Activity 完整流程伪代码**

```java
// ATMS.java - 前台导演
class ActivityTaskManagerService {
    // 持有 AMS 引用
    ActivityManagerService mActivityManager;
    
    // 启动Activity的主入口
    Result startActivity(Intent intent, Bundle options) {
        // 1. 解析Intent和启动标志
        ActivityInfo activityInfo = resolveActivity(intent);
        int launchFlags = getLaunchFlags(intent);
        
        // 2. 确定目标任务栈策略
        Task targetTask = getTargetTask(intent, activityInfo);
        
        // 3. 创建或获取Activity记录
        ActivityRecord r = new ActivityRecord(activityInfo);
        r.task = targetTask;
        
        // 4. 检查权限（委托给AMS）
        //    ↓↓↓ 关键交互点1：权限检查 ↓↓↓
        if (!mActivityManager.checkPermission(r, intent)) {
            throw SecurityException("Permission denied");
        }
        
        // 5. 获取或创建目标进程
        //    ↓↓↓ 关键交互点2：进程管理 ↓↓↓
        ProcessRecord targetProcess = mActivityManager.getProcessForActivity(r);
        if (targetProcess == null) {
            // 进程不存在，需要创建
            targetProcess = mActivityManager.startProcess(
                r.info.applicationInfo,
                r.info.processName
            );
        }
        
        // 6. 设置进程优先级（因为即将成为前台进程）
        mActivityManager.updateProcessPriority(
            targetProcess, 
            FOREGROUND_APP_ADJ
        );
        
        // 7. 暂停当前Activity
        ActivityRecord prev = getResumedActivity();
        if (prev != null) {
            prev.state = PAUSING;
            sendPauseToClient(prev);
        }
        
        // 8. 启动目标Activity
        r.state = STARTING;
        targetProcess.scheduleLaunchActivity(r);
        
        // 9. 与WMS协调窗口
        WindowManagerService wms = getWindowManager();
        wms.addWindow(r.token, r);
        
        return SUCCESS;
    }
}
```

```java
// AMS.java - 后台大总管
class ActivityManagerService {
    // 持有 ATMS 引用
    ActivityTaskManagerService mActivityTaskManager;
    
    // 所有进程记录
    Map<String, ProcessRecord> mProcessNames;
    Map<Integer, ProcessRecord> mProcessPids;
    
    // 权限检查
    boolean checkPermission(ActivityRecord r, Intent intent) {
        // 1. 检查包名权限
        if (!checkPackagePermission(r.packageName, callingUid)) {
            return false;
        }
        
        // 2. 检查特定组件权限
        if (r.info.permission != null) {
            return checkComponentPermission(r.info.permission, callingPid, callingUid);
        }
        
        // 3. URI权限检查
        if (intent.getData() != null) {
            return checkUriPermission(intent.getData(), callingUid);
        }
        
        return true;
    }
    
    // 获取或创建进程
    ProcessRecord getProcessForActivity(ActivityRecord r) {
        String processName = r.info.processName;
        
        // 1. 查找已存在的进程
        ProcessRecord app = mProcessNames.get(processName);
        if (app != null && app.thread != null) {
            // 进程存在且活跃
            return app;
        }
        
        // 2. 需要创建新进程
        return startProcessLocked(
            r.info.applicationInfo,
            processName,
            r.info.packageName
        );
    }
    
    // 启动新进程
    ProcessRecord startProcessLocked(ApplicationInfo info, String processName) {
        // 1. 创建进程记录
        ProcessRecord app = new ProcessRecord(info, processName);
        mProcessNames.put(processName, app);
        
        // 2. 通过Zygote fork新进程
        int pid = Process.startViaZygote(...);
        app.pid = pid;
        mProcessPids.put(pid, app);
        
        // 3. 建立与应用进程的Binder连接
        // （应用进程启动后会调用attachApplication）
        
        return app;
    }
    
    // 更新进程优先级（OOM_ADJ）
    void updateProcessPriority(ProcessRecord app, int adj) {
        app.curAdj = adj;
        
        // 1. 更新内核OOM调整值
        writeOomAdjToProc(app.pid, adj);
        
        // 2. 通知相关组件
        if (adj >= CACHED_APP_MIN_ADJ) {
            // 进入缓存状态，可能被回收
            scheduleTrimMemory(app, TRIM_MEMORY_BACKGROUND);
        }
    }
}
```

## **3. ActivityRecord 和 ProcessRecord 数据结构伪代码**

```java
// ATMS管理的核心数据结构
class ActivityRecord {
    // 基本标识
    ActivityInfo info;           // AndroidManifest中的信息
    IBinder token;              // 唯一标识符，与WMS通信使用
    Task task;                  // 所属的任务
    
    // 生命周期状态（ATMS视角）
    int mState;                 // INITIALIZING, RESUMED, PAUSED, STOPPED, DESTROYED
    
    // 指向AMS管理的进程
    ProcessRecord app;          // 所属的进程
    
    // 窗口相关
    WindowContainer windowContainer;
    
    // 状态变化方法
    void setState(int newState) {
        mState = newState;
        
        // 通知AMS更新进程状态
        if (app != null) {
            app.updateActivityState(this);
            
            // 如果变成前台Activity，提升进程优先级
            if (newState == RESUMED) {
                mActivityManager.updateProcessPriority(app, FOREGROUND_APP_ADJ);
            }
        }
    }
}

// AMS管理的核心数据结构
class ProcessRecord {
    // 进程标识
    int pid;
    String processName;
    ApplicationInfo info;
    
    // 进程优先级
    int curAdj;                // 当前OOM调整值（-1000~1000）
    int setAdj;                // 上次设置的值
    
    // 与ATMS的关联
    List<ActivityRecord> activities = new ArrayList<>();
    
    // Binder连接
    IApplicationThread thread; // 与应用进程通信的接口
    
    // 进程状态
    boolean isForeground = false;
    
    // 添加Activity（由ATMS调用）
    void addActivity(ActivityRecord r) {
        activities.add(r);
        
        // 更新进程状态
        updateForegroundState();
    }
    
    // 移除Activity（由ATMS调用）
    void removeActivity(ActivityRecord r) {
        activities.remove(r);
        updateForegroundState();
    }
    
    // 更新前台状态
    void updateForegroundState() {
        boolean hasForegroundActivity = false;
        
        for (ActivityRecord r : activities) {
            if (r.mState == RESUMED) {
                hasForegroundActivity = true;
                break;
            }
        }
        
        isForeground = hasForegroundActivity;
    }
}
```

```java
// 进入分屏模式
class MultiWindowManager {
    // 用户拖动Activity到边缘
    void onDragToSplit(ActivityRecord draggedActivity) {
        // 1. ATMS重新组织任务栈
        TaskStack leftStack = mActivityTaskManager.createStack(
            WINDOWING_MODE_SPLIT_SCREEN_PRIMARY
        );
        TaskStack rightStack = mActivityTaskManager.createStack(
            WINDOWING_MODE_SPLIT_SCREEN_SECONDARY
        );
        
        // 2. 移动Activity到新栈
        leftStack.addActivity(draggedActivity);
        
        // 3. 找到另一个Activity显示在右侧
        ActivityRecord otherActivity = findActivityForSplit();
        rightStack.addActivity(otherActivity);
        
        // 4. 通知AMS更新进程优先级
        //    ↓↓↓ 关键交互点：多窗口优先级调整 ↓↓↓
        mActivityManager.updateMultiWindowPriority(
            draggedActivity.app,    // 左侧进程
            otherActivity.app,      // 右侧进程
            leftStack,
            rightStack
        );
        
        // 5. 协调WMS布局
        mWindowManager.setSplitScreenBounds(
            leftStack.bounds,
            rightStack.bounds
        );
    }
}

// AMS中的多窗口优先级处理
class ActivityManagerService {
    void updateMultiWindowPriority(ProcessRecord leftApp, 
                                   ProcessRecord rightApp,
                                   TaskStack leftStack,
                                   TaskStack rightStack) {
        // 1. 左侧栈的进程（主栈）获得更高优先级
        leftApp.curAdj = FOREGROUND_APP_ADJ - 1;  // 比普通前台略高
        writeOomAdjToProc(leftApp.pid, leftApp.curAdj);
        
        // 2. 右侧栈的进程（次栈）中等优先级
        rightApp.curAdj = VISIBLE_APP_ADJ;  // 可见但非焦点
        writeOomAdjToProc(rightApp.pid, rightApp.curAdj);
        
        // 3. 其他后台进程降低优先级
        for (ProcessRecord app : getAllProcesses()) {
            if (app != leftApp && app != rightApp && app.isBackground()) {
                app.curAdj = CACHED_APP_ADJ;
                writeOomAdjToProc(app.pid, app.curAdj);
            }
        }
    }
}
```

```java
// InputDispatcher检测到超时
class InputDispatcher {
    void onInputEventTimeout(WindowHandle window) {
        // 1. 找到负责的进程
        //    ↓↓↓ 关键交互点1：确定责任进程 ↓↓↓
        ActivityRecord activity = mActivityTaskManager.findActivityByToken(window.token);
        
        if (activity != null) {
            ProcessRecord app = activity.app;
            
            // 2. 报告给AMS
            //    ↓↓↓ 关键交互点2：ANR处理 ↓↓↓
            mActivityManager.appNotResponding(
                app,                     // 问题进程
                activity,                // 问题Activity
                "Input dispatching timed out",  // 原因
                window.getInputChannel() // 输入通道
            );
        }
    }
}

// AMS处理ANR
class ActivityManagerService {
    void appNotResponding(ProcessRecord app, 
                          ActivityRecord activity,
                          String reason,
                          Object inputChannel) {
        // 1. 收集ANR信息
        ANRInfo anrInfo = new ANRInfo();
        anrInfo.process = app;
        anrInfo.activity = activity;
        anrInfo.reason = reason;
        
        // 2. 获取当前堆栈信息（需要ATMS提供）
        //    ↓↓↓ 关键交互点3：获取前台信息 ↓↓↓
        anrInfo.focusedStack = mActivityTaskManager.getFocusedStack();
        anrInfo.lastResumedActivity = mActivityTaskManager.getLastResumedActivity();
        
        // 3. 判断是否弹出ANR对话框
        boolean shouldShowDialog = true;
        
        // 检查：如果ANR发生在后台，可能不弹框
        if (activity == null || activity.mState != RESUMED) {
            shouldShowDialog = false;
        }
        
        // 4. 记录日志
        logANR(anrInfo);
        
        // 5. 弹出对话框（如果在前台）
        if (shouldShowDialog) {
            showANRDialog(app, activity, anrInfo);
        } else {
            // 后台ANR，直接杀进程
            killProcessGroup(app.pid);
            cleanupProcess(app);
            
            // 通知ATMS清理Activity
            mActivityTaskManager.cleanupActivitiesForProcess(app);
        }
    }
}
```

```java
// 当系统内存不足时
class LowMemoryKiller {
    void onLowMemory() {
        // 1. 获取所有进程按优先级排序
        List<ProcessRecord> processes = mActivityManager.getAllProcessesByAdj();
        
        for (ProcessRecord app : processes) {
            // 2. 跳过关键系统进程
            if (app.isSystem()) continue;
            if (app.isPersistent()) continue;
            
            // 3. 检查是否有前台Activity（需要ATMS信息）
            //    ↓↓↓ 关键交互点：检查前台状态 ↓↓↓
            boolean hasForegroundActivity = 
                mActivityTaskManager.hasForegroundActivity(app);
            
            if (!hasForegroundActivity) {
                // 4. 可以杀死这个进程
                int pid = app.pid;
                
                // 5. 通知ATMS清理Activity
                mActivityTaskManager.onProcessKilled(app);
                
                // 6. 执行杀进程
                killProcess(pid);
                
                // 7. AMS清理内部状态
                mActivityManager.cleanupProcessLocked(app);
                
                // 检查是否已释放足够内存
                if (checkMemoryImproved()) {
                    break;
                }
            }
        }
    }
}
```

## **8. 总结：核心交互模式伪代码**

```java
// 最简化的交互模式
class ATMS_AMS_Interaction {
    
    // 模式1：ATMS委托AMS处理"后台事务"
    void pattern_DelegateToAMS(ActivityRecord r) {
        // ATMS说："我需要一个进程来运行这个Activity"
        ProcessRecord app = mAMS.getOrCreateProcess(r);
        
        // ATMS说："这个进程应该有前台优先级"
        mAMS.updateProcessPriority(app, FOREGROUND);
        
        // ATMS说："检查是否有权限"
        if (!mAMS.checkPermission(r)) {
            abort("No permission");
        }
    }
    
    // 模式2：AMS咨询ATMS处理"前台状态"
    void pattern_QueryFromATMS(ProcessRecord app) {
        // AMS说："我要杀这个进程，它有前台Activity吗？"
        if (mATMS.hasForegroundActivity(app)) {
            // 有前台Activity，谨慎处理
            deferKilling(app);
        } else {
            // 没有前台Activity，可以杀
            killProcess(app);
            mATMS.cleanupActivities(app);
        }
        
        // AMS说："发生ANR了，哪个Activity是焦点？"
        ActivityRecord focused = mATMS.getFocusedActivity();
        if (focused != null && focused.app == app) {
            showANRDialog(focused);
        }
    }
    
    // 模式3：双向状态同步
    void pattern_StateSync() {
        // 当Activity状态变化时
        onActivityStateChanged(ActivityRecord r, int newState) {
            // ATMS更新Activity状态
            r.setState(newState);
            
            // 通知AMS更新进程状态
            r.app.updateActivityState(r);
        }
        
        // 当进程状态变化时
        onProcessStateChanged(ProcessRecord app, int newAdj) {
            // AMS更新进程优先级
            app.setAdj(newAdj);
            
            // 可能影响Activity调度
            if (newAdj > CACHED_THRESHOLD) {
                // 进程进入缓存，考虑暂停其Activity
                for (ActivityRecord r : app.activities) {
                    mATMS.schedulePauseIfNeeded(r);
                }
            }
        }
    }
}
```

这些伪代码展示了ATMS和AMS如何通过**清晰的接口契约**、**共享的数据结构**和**同步的锁机制**协同工作，共同管理Android应用的复杂生命周期和资源调度。
