import android.view.KeyEvent  
import android.view.ViewGroup  
import android.view.inputmethod.EditorInfo  
import android.widget.EditText  
import androidx.lifecycle.MutableLiveData  
import com.chad.library.adapter.base.BaseQuickAdapter  
import com.chad.library.adapter.base.viewholder.BaseViewHolder  
import com.yto.pda.cwms.R  
import com.yto.pda.cwms.data.model.bean.BatchReviewGoodsList  
import com.yto.pda.cwms.network.stateCallback.UiState  
import com.yto.pda.cwms.util.CacheUtil  
  
class BatchReviewDetailAdapter(data: MutableList<BatchReviewGoodsList>) :  
  
    BaseQuickAdapter<BatchReviewGoodsList, BaseViewHolder>(R.layout.item_last_review_detail, data) {  
  
    var flag = false  
  
    var refresh: MutableLiveData<UiState> = MutableLiveData()  
  
    //20250107 新加逻辑，数量是否可以手动编辑  
    var isEditNum = false  
  
    init {  
        CacheUtil.getUser()?.let {  
            flag = it.pickPictureFlag  
        }  
    }  
  
    fun isEditNum(isEditNum: Boolean) {  
        this.isEditNum = isEditNum  
    }  
  
	/**
	替换布局 生成不同的viewHolder
	**/
    override fun onCreateDefViewHolder(parent: ViewGroup, viewType: Int): BaseViewHolder {  
  
        if (flag){  
            return createBaseViewHolder(parent, R.layout.item_last_review_detail)  
        }else{  
            return createBaseViewHolder(parent, R.layout.item_last_review_detail_no_pic)  
        }  
  
  
    }  
  
  
    override fun convert(holder: BaseViewHolder, item: BatchReviewGoodsList) {  
  
        holder.setText(R.id.tv_shop_name, item.itemName)  
  
        holder.setText(R.id.tv_shop_barcode, item.barCode)  
  
        holder.setText(R.id.tv_shop_code, item.itemCode)  
  
        holder.setText(R.id.tv_excepted_num, item.expectQty.toString())  
  
        holder.setText(R.id.tv_shop_type, item.skuProperty)  
  
        if (item.scaned > 0) {  
            holder.setText(R.id.et_scaned_num, item.scaned.toString())  
        }  
  
  
        holder.setBackgroundResource(  
            R.id.item_root_view,  
            if (item.isSelected) R.drawable.bg_stroke_5 else 0  
        )  
        val editActualPick = holder.getView<EditText>(R.id.et_scaned_num)  
        editActualPick.isEnabled = isEditNum  
        editActualPick.setOnEditorActionListener { view, actionId, event ->  
            if (actionId == EditorInfo.IME_ACTION_SEND || actionId == EditorInfo.IME_ACTION_DONE || event != null  
                && KeyEvent.KEYCODE_ENTER == event.keyCode && KeyEvent.ACTION_DOWN == event.action  
            ) {  
                var et = view as EditText  
                item.scaned = et.text.toString().toInt()  
                if (item.scaned >= item.expectQty) {  
                    item.scaned = item.expectQty  
                    et.setText(item.scaned.toString())  
                }  
                et.clearFocus()  
  
                val uiState = UiState(isSuccess = true, "", successMsg = "")  
                refresh.value = uiState  
            }  
            true  
        }  
  
    }  
  
  
}
